# Alocação contígua: fragmentação interna, externa e compactação
format --block-size 1KB --blocks 16 --alloc contiguous
create /a.dat --size 3KB
create /b.dat --size 2500
create /c.dat --size 2KB
create /d.dat --size 1500
# remover b.dat abre um "buraco" no meio do disco
rm /b.dat
fs_status --details

compact
